{{- /* Embeds another page's card inline in the body, rendering through page/card.html (preset inline) instead of the deleted article/embedded partial. Parameters link (string) — target page path ("/journal/x/"); empty means the current page; an unknown path warns in hugo server and renders nothing (the source template crashed on it) showSummary (bool) — summary visibility; falls back to the target's front matter, then site.Params.list.showSummary, then false. Parsed via explicit "true"/"false" comparison so a string "false" cannot read as truthy compactSummary is not read by this shortcode, so the smoke whitelist rejects it; the card CSS clamps the summary to 3 lines. The embedded card never loads eagerly — body images are always lazy. Called by: content markdown */ -}} {{- $target := .Page -}} {{- $link := .Get "link" -}} {{- if $link -}} {{- with .Site.GetPage $link -}} {{- $target = . -}} {{- else -}} {{- if hugo.IsServer -}} {{- warnf "content/article: no page matches link=%q (%s)" $link $.Position -}} {{- end -}} {{- $target = false -}} {{- end -}} {{- end -}} {{- if $target -}} {{- $show := false -}} {{- $raw := printf "%v" (.Get "showSummary") -}} {{- if eq $raw "true" -}} {{- $show = true -}} {{- else if ne $raw "false" -}} {{- $show = site.Params.list.showSummary -}} {{- if isset $target.Params "showsummary" -}}{{- $show = $target.Params.showsummary -}}{{- end -}} {{- end -}}
{{- partial "page/card.html" (dict "page" $target "preset" "inline" "showSummary" $show "eager" false) -}}
{{- end -}}
