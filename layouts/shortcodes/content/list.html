{{- /* Lists recent regular pages as cards inside an article, rendering through page/card.html. Parameters limit (int) — how many pages to consider, default 1 title (string) — heading above the list; default is the i18n key "shortcode.recent_articles" with an English fallback (no catalogue ships yet) where (string) — page field to filter on (e.g. "Type") value (any) — the value the field must equal cardView (bool) — true renders grid cards, false or absent renders row cards; compared against an explicit "true" so a string "false" selects the row branch (the source template's truthy check sent it to the grid branch) summary (bool) — show each page's summary on its card, default true; compared against an explicit "false" so a string "true" still shows it The page containing the shortcode never appears in its own list. Called by: content markdown */ -}} {{- $limit := int (default "1" (.Get "limit")) -}} {{- $title := .Get "title" | default (i18n "shortcode.recent_articles") | default "Recent articles" -}} {{- $where := .Get "where" -}} {{- $value := .Get "value" -}} {{- $cardView := eq (printf "%v" (.Get "cardView")) "true" -}} {{- $showSummary := ne (printf "%v" (.Get "summary")) "false" -}} {{- $preset := cond $cardView "grid" "row" -}} {{- $pages := .Site.RegularPages -}} {{- if $where }}{{- $pages = where $pages $where $value -}}{{- end -}} {{- $parent := .Page.RelPermalink -}}

{{ $title }}

{{- range $i, $p := first $limit $pages }} {{- if ne $p.RelPermalink $parent }} {{- partial "page/card.html" (dict "page" $p "preset" $preset "showSummary" $showSummary "index" $i) }} {{- end }} {{- end }}
