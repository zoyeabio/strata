{{- /* Site header: brand plus main navigation. Brand rules - A resolvable `site.Params.logo` (an asset path) is rendered instead of the plain text title. - `disableTextInHeader` only hides the text when a logo is present. Without a logo the title always renders, so a site that forgets the logo never ends up with an empty header. The `header.layout` shell (smart / standard) is not wired up yet; it needs the scroll behaviour script that belongs with it. Called by: _default/baseof.html */ -}} {{- $logo := "" }} {{- $hasLogo := false }} {{- with site.Params.logo }} {{- with resources.Get . }} {{- $logo = . }} {{- $hasLogo = true }} {{- end }} {{- end }} {{- $disableText := site.Params.disableTextInHeader | default false }} {{- $showTitle := or (not $hasLogo) (not $disableText) -}}
{{- with $logo }} {{- if eq .MediaType.SubType "svg" }} {{ .Content | safeHTML }} {{- else }} {{- end }} {{ site.Title }} {{- end }} {{- if $showTitle }} {{ site.Title }} {{- end }} {{ partial "frame/menu.html" . }}
