{{- /* Resolves one image reference into responsive WebP variants. Returns a dict, never HTML. Input dict ref (string, required) — front matter value or markdown destination page (Page, optional) — page bundle lookups and section fallbacks role (string, optional) — "card" | "illustration" | "hero" | "og"; caps the widest variant alt (string, optional) — empty string means decorative quality (int, optional) — WebP quality, defaults to [params.image].quality; the figure shortcode's quality parameter passes it through Output dict ok (bool) — false when nothing resolves; the caller renders nothing url (string) — the largest variant, or the untouched original file srcset (string) — "" when one file is enough width (int) — 0 when unknown (svg, passthrough) height (int) — 0 when unknown (svg, passthrough) alt (string) — echoed from the input Resolution order: page resource → the reference as written → assets path under images/{pageDir}/ (the article's own directory, so a section-flat assets/images/{section}/file.png and a per-page assets/images/{section}/{slug}/file.png both resolve) → images/ → static/ → not found: a declared reference that resolves to nothing fails production builds (errorf) and warns in `hugo server`; an empty reference is not an error. http(s) references are passed through only when site.Params.allowRemoteImages is set; otherwise they are skipped with the same dev-only warning. The variant ladder, role caps, WebP quality and social preview size are config — [params.image] in the theme's root config.toml, which a site overrides in its own. Each ladder step is capped by the source width and by the role cap, then de-duplicated, then a step less than 15% wider than its kept predecessor is dropped. The result is never more than four files per image and never two near-identical widths — the largest variant may sit a little below the source width, which is accepted in exchange for the smaller payload. Everything is encoded at one quality so the same width is never written twice. SVG, GIF and animated WebP bypass processing entirely — the imaging pipeline would fail on SVG and flatten animations to frame one. site.Params.disableImageOptimization = true hands back the original file for every role, generating no variants at all. Called by: page/card.html, opengraph.html, _markup/render-image.html */ -}} {{- $raw := printf "%v" .ref }} {{- if eq $raw "" }}{{ $raw = "" }}{{ end }} {{- $ref := trim $raw " /" }} {{- $page := .page }} {{- $role := .role | default "illustration" -}} {{- /* Pipeline knobs come from [params.image] in the theme's root config.toml, which a site overrides in its own. The fallbacks mirror the theme's defaults, so a site written before those keys existed still renders. */ -}} {{- $imgCfg := site.Params.image | default dict -}} {{- $capsCfg := $imgCfg.caps | default dict -}} {{- $ogCfg := $imgCfg.og | default dict -}} {{- $caps := dict "card" (int ($capsCfg.card | default 800)) "illustration" (int ($capsCfg.illustration | default 1280)) "hero" (int ($capsCfg.hero | default 1920)) "og" (int ($capsCfg.og | default 1920)) -}} {{- $cap := index $caps $role | default 1280 -}} {{- $ladder := slice -}} {{- range $imgCfg.ladder | default (slice 480 800 1280 1920) -}} {{- $ladder = $ladder | append (int .) -}} {{- end -}} {{- $quality := .quality | default (int ($imgCfg.quality | default 78)) -}} {{- $ogWidth := int ($ogCfg.width | default 1200) -}} {{- $ogHeight := int ($ogCfg.height | default 630) -}} {{- $disable := site.Params.disableImageOptimization | default false }} {{- $res := "" }} {{- $remote := false }} {{- if and $ref (or (strings.HasPrefix $ref "http://") (strings.HasPrefix $ref "https://")) }} {{- $remote = true }} {{- else if $ref }} {{- with $page }}{{ with .Resources.Get $ref }}{{ $res = . }}{{ end }}{{ end }} {{- if not $res }}{{ with resources.Get $ref }}{{ $res = . }}{{ end }}{{ end }} {{- if not $res }} {{- $pageDir := "" }} {{- with $page }}{{ with .File }}{{ $pageDir = path.Dir .Path }}{{ end }}{{ end }} {{- $candidates := slice }} {{- if $pageDir }}{{ $candidates = $candidates | append (printf "images/%s/%s" $pageDir $ref) }}{{ end }} {{- $candidates = $candidates | append (printf "images/%s" $ref) }} {{- range $candidates }}{{ if not $res }}{{ with resources.Get . }}{{ $res = . }}{{ end }}{{ end }}{{ end }} {{- end }} {{- end }} {{- if and $ref (not $res) (not $remote) }} {{- if fileExists (printf "static/%s" $ref) }} {{- $res = dict "RelPermalink" (relURL $ref) "ResourceType" "file" }} {{- end }} {{- end }} {{- $result := dict "ok" false "url" "" "srcset" "" "width" 0 "height" 0 }} {{- if $remote }} {{- if site.Params.allowRemoteImages }} {{- $result = dict "ok" true "url" $ref "srcset" "" "width" 0 "height" 0 }} {{- else if hugo.IsServer }} {{- warnf "strata: remote image skipped, set site.Params.allowRemoteImages to allow it (ref=%q role=%q)" $ref $role }} {{- end }} {{- else if $res }} {{- if and (eq $res.ResourceType "image") (not $disable) }} {{- $subtype := $res.MediaType.SubType }} {{- $animated := false }} {{- if eq $subtype "webp" }} {{- $sniff := slice (printf "assets/%s" $res.Name) }} {{- with $page }}{{ with .File }}{{ $sniff = $sniff | append (printf "content/%s/%s" (path.Dir .Path) $res.Name) }}{{ end }}{{ end }} {{- range $sniff }}{{ if not $animated }}{{ if fileExists . }}{{ $animated = strings.Contains (readFile .) "ANIM" }}{{ end }}{{ end }}{{ end }} {{- end }} {{- if or (eq $subtype "svg") (eq $subtype "gif") $animated }} {{- $result = dict "ok" true "url" $res.RelPermalink "srcset" "" "width" 0 "height" 0 }} {{- else if eq $role "og" }} {{- $social := $res.Fill (printf "%dx%d jpeg q%d" $ogWidth $ogHeight $quality) }} {{- $result = dict "ok" true "url" $social.RelPermalink "srcset" "" "width" $social.Width "height" $social.Height }} {{- else }} {{- $sourceWidth := $res.Width }} {{- $max := int (math.Min $sourceWidth $cap) }} {{- $steps := slice }} {{- range $ladder }} {{- $step := int (math.Min . $max) }} {{- if not (in $steps $step) }}{{ $steps = $steps | append $step }}{{ end }} {{- end }} {{- $kept := slice }} {{- range $steps }} {{- if eq (len $kept) 0 }} {{- $kept = $kept | append . }} {{- else }} {{- $prev := index $kept (sub (len $kept) 1) }} {{- if gt (mul (sub . $prev) 100) (mul $prev 15) }} {{- $kept = $kept | append . }} {{- end }} {{- end }} {{- end }} {{- $parts := slice }} {{- $last := "" }} {{- range $kept }} {{- $variant := $res.Resize (printf "%dx webp q%d" . $quality) }} {{- $parts = $parts | append (printf "%s %dw" $variant.RelPermalink .) }} {{- $last = $variant }} {{- end }} {{- $result = dict "ok" true "url" $last.RelPermalink "srcset" (delimit $parts ", ") "width" $last.Width "height" $last.Height }} {{- end }} {{- else }} {{- $result = dict "ok" true "url" $res.RelPermalink "srcset" "" "width" 0 "height" 0 }} {{- end }} {{- else if $ref }} {{- $where := "" }}{{ with $page }}{{ $where = .RelPermalink }}{{ end }} {{- if hugo.IsProduction }} {{- errorf "strata: image not found (ref=%q role=%q page=%q)" $ref $role $where }} {{- else if hugo.IsServer }} {{- warnf "strata: image not found (ref=%q role=%q page=%q)" $ref $role $where }} {{- end }} {{- end }} {{- return (merge $result (dict "alt" (.alt | default ""))) }}
