{{- /* Partial: data/image-ref.html Role: Resolve the image reference a page renders. Cards, term cards and heroes take featureimage, else a legacy bundle resource whose name contains feature, cover or thumbnail. Social output takes params.images[0] instead of the site default, and ends in site.Params.defaultSocialImage; featureimage still wins over images so the picture a page shows is the picture it advertises. Params: page (Page, required) social (bool, optional) — social order Outputs: string (asset ref or URL, may be empty) Called by: page/card.html, page/term-card.html, data/hero-image.html, opengraph.html, head/schema.html */ -}} {{- $page := .page }} {{- $social := .social | default false }} {{- $ref := "" }} {{- if $social }}{{ with $page.Params.images }}{{ $ref = index . 0 }}{{ end }}{{ end }} {{- with $page.Params.featureimage }}{{ $ref = . }}{{ end }} {{- if not $ref }} {{- range $page.Resources }} {{- if and (not $ref) (or (in .Name "feature") (in .Name "cover") (in .Name "thumbnail")) }} {{- $ref = .Name }} {{- end }} {{- end }} {{- end }} {{- if $social }}{{ with site.Params.defaultSocialImage }}{{ if not $ref }}{{ $ref = . }}{{ end }}{{ end }}{{ end }} {{- return $ref }}
