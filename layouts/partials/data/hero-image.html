{{- /* Returns the hero image dict for a page: the resolver result plus its `sizes` string. Pages without a hero get a dict with ok=false. A regular page takes its image from data/image-ref.html (featureimage, else a bundle resource named feature/cover/thumbnail) unless the page says showHero: false. A section page takes one only when [list] showHero is true, so a section banner is opt-in. Either way a page-level showHero wins over the table. The result is cached on the page scratch so that head/head.html (which preloads the hero) and page/hero.html render the exact same srcset from a single computation — two independent calls are how preload and drift. Called by: head/head.html, _default/single.html, _default/list.html */ -}} {{- $page := . }} {{- $img := $page.Scratch.Get "strataHeroImage" }} {{- if not $img }} {{- $img = dict "ok" false }} {{- $hero := false }} {{- if isset $page.Params "showhero" }}{{ $hero = $page.Params.showhero }} {{- else if $page.IsPage }}{{ $hero = true }} {{- else if eq $page.Kind "section" }}{{ $hero = site.Params.list.showHero }}{{ end }} {{- if $hero }} {{- with (partial "data/image-ref.html" (dict "page" $page)) }} {{- $img = partial "data/image-resource.html" (dict "ref" . "page" $page "role" "hero" "alt" ($page.Params.featureimagealt | default (printf "Featured image for %s" $page.Title))) }} {{- $img = merge $img (dict "sizes" (partial "data/card-presets.html" "hero").sizes) }} {{- end }} {{- end }} {{- $page.Scratch.Set "strataHeroImage" $img }} {{- end }} {{- return $img }}
