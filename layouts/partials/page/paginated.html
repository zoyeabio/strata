{{- /* Sort a page's children, paginate them, render the collection and the pager. Params page (Page, required) — the section or term page params (map, required) — site.Params.list or site.Params.term; the page's own front matter wins over every default in it Called by: _default/list.html, _default/term.html */ -}} {{- $page := .page }} {{- $params := .params }} {{- $sorted := $page.Pages }} {{- $order := $params.orderByWeight }}{{ if isset $page.Params "orderbyweight" }}{{ $order = $page.Params.orderbyweight }}{{ end }} {{- if $order }} {{- $sorted = $page.Pages.ByWeight }} {{- end }} {{- $cardView := $params.cardView }}{{ if isset $page.Params "cardview" }}{{ $cardView = $page.Params.cardview }}{{ end }} {{- $showSummary := $params.showSummary }}{{ if isset $page.Params "showsummary" }}{{ $showSummary = $page.Params.showsummary }}{{ end }} {{- $pager := $page.Paginate $sorted }} {{ partial "page/page-collection.html" (dict "pages" $pager.Pages "cardView" $cardView "showSummary" $showSummary ) }} {{ partial "page/pagination.html" $pager }}
